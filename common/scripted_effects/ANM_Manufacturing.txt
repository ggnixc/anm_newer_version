manufacture_recalculate_share_percentages = { #no longer necessary
	set_temp_variable = { coolandfunnyanmvariable = 0 }
	set_temp_variable = { temporaryfunnything = 0 }
	for_loop_effect = {
			start = 0
			end = 13
			add_to_temp_variable = { temporaryfunnything = anm_manufacture_share^coolandfunnyanmvariable }
			add_to_temp_variable = { coolandfunnyanmvariable = 1 }
			
	}
	set_variable = { debugggggggg = temporaryfunnything  }
	set_temp_variable = { coolandfunnyanmvariable = 0 }
	set_temp_variable = { adduptohundred = 100 }
	for_loop_effect = {
			start = 0
			end = 13
			set_temp_variable = { lefunnyandcoolanmtempvar = anm_manufacture_share^coolandfunnyanmvariable }
			multiply_temp_variable = { lefunnyandcoolanmtempvar = 100 }
			divide_temp_variable = { lefunnyandcoolanmtempvar = temporaryfunnything }
			round_temp_variable = lefunnyandcoolanmtempvar 
			#add_to_temp_variable = { lefunnyandcoolanmtempvar = -1 }
			subtract_from_temp_variable = { adduptohundred = lefunnyandcoolanmtempvar }
			set_variable = { anm_manufacture_share^coolandfunnyanmvariable = lefunnyandcoolanmtempvar }

			add_to_temp_variable = { coolandfunnyanmvariable = 1 }
			
	}
	set_temp_variable = { coolandfunnyanmvariable = 0 }
	for_loop_effect = {
			start = 0
			end = 13
			set_temp_variable = { lefunnyandcoolanmtempvar = anm_manufacture_share^coolandfunnyanmvariable }
			#total rest value higher than 0
			if = {
				limit = {
				check_variable = { adduptohundred > 0 }
				}
				add_to_temp_variable = { lefunnyandcoolanmtempvar = 1 }
				subtract_from_temp_variable = { adduptohundred = -1 }
			}
			#total value lower than 0
			if = {
				limit = {
				check_variable = { 0 > adduptohundred }
				check_variable = { lefunnyandcoolanmtempvar > 0 }
				}
				add_to_temp_variable = { lefunnyandcoolanmtempvar = -1 }
				add_to_temp_variable = { adduptohundred = 1 }
				
			}
			
			set_variable = { anm_manufacture_share^coolandfunnyanmvariable = lefunnyandcoolanmtempvar }

			add_to_temp_variable = { coolandfunnyanmvariable = 1 }
			
	}
}
