### Party/Piechart related scripts ###
ANM_USA_Calculate_Party_Balance = {
	set_temp_variable = { USA_Party_Dem_Total = USA_Party_Dem_Labour }
		add_to_temp_variable = { USA_Party_Dem_Total = USA_Party_Dem_Center }
		add_to_temp_variable = { USA_Party_Dem_Total = USA_Party_Dem_Innovation }
	set_temp_variable = { USA_Party_Rep_Total = USA_Party_Rep_Responsible }
		add_to_temp_variable = { USA_Party_Rep_Total = USA_Party_Rep_Consensus }
		add_to_temp_variable = { USA_Party_Rep_Total = USA_Party_Rep_Freedom }

	set_temp_variable = { USA_Party_Balance = USA_Party_Dem_Total }
		subtract_from_temp_variable = { USA_Party_Balance = USA_Party_Rep_Total } 
	# Negative = Republican Dominance, Positive = Democrat Dominance
}
