state_control_effect_2030 = { 
	if = {
		limit = { is_controlled_by = AAG }
		set_state_name = STATE_2030_aag_eng
		reset_province_name = 17785
		if = {
			limit = { has_global_flag = TNO_endonym_mode }
			set_state_name = STATE_2030_aag
		}
	}
	else = {
		reset_state_name = yes
		set_province_name = { id = 17785 name = VICTORY_POINTS_17785_aao }
	}
}
