# Initialize sphere size impact on US GDP Growth
USA_econ_sphere_init = {
    set_variable = { USA_ECON_SPHERE_OFFSET = 1560 } # Sphere must be at least 1.30 times the size of the US's economy to provide benefit (times 1200 to minimize truncation)
    set_variable = { USA_ECON_SPHERE_SCALE = 2 }

    USA_econ_sphere_calc_value = yes
    log = "[?temp_ratio]"
    clear_array = USA_econ_sphere_ratio_array
    resize_array = {
        array = USA_econ_sphere_ratio_array
        value = temp_ratio
        size = 12
    }
    USA_econ_sphere_recalc = yes
}

USA_econ_sphere_tick = {
    remove_from_array = {
        array = USA_econ_sphere_ratio_array
        index = 0
    }
    USA_econ_sphere_calc_value = yes
    add_to_array = { USA_econ_sphere_ratio_array = temp_ratio }
    USA_econ_sphere_recalc = yes
}

USA_econ_sphere_calc_value = {
    set_temp_variable = { temp_ratio = total_econ_sphere_GDP }
    multiply_temp_variable = { temp_ratio = 100 }
    divide_temp_variable = { temp_ratio = GDP }
}

USA_econ_sphere_recalc = {
    set_variable = { USA_NWM_inflation = 0 }
    set_variable = { USA_NWM_gdp = 0 }
    set_temp_variable = { USA_econ_sphere_sum = 0 }
    for_each_loop = {
        array = USA_econ_sphere_ratio_array
        value = v

        add_to_temp_variable = { USA_econ_sphere_sum = v }
    }
    subtract_from_temp_variable = { USA_econ_sphere_sum = USA_ECON_SPHERE_OFFSET }
    divide_temp_variable = { USA_econ_sphere_sum = 12 }
    set_variable = { USA_econ_sphere_modifier = USA_econ_sphere_sum }
    multiply_variable = { USA_econ_sphere_modifier = USA_ECON_SPHERE_SCALE }
    divide_variable = { USA_econ_sphere_modifier = 100 }

    if = {
        limit = {
            check_variable = { USA_econ_sphere_modifier < 0 }
        }
        set_temp_variable = { inflation_penalty = USA_econ_sphere_modifier }
        multiply_temp_variable = { inflation_penalty = -0.25 }
        set_variable = { USA_NWM_inflation = inflation_penalty}
    }
    if = {
        limit = {
            check_variable = { USA_econ_sphere_modifier < -0.15 }
        }
        set_temp_variable = { gdp_penalty = USA_econ_sphere_modifier }
        multiply_temp_variable = { gdp_penalty = 0.17 }
        set_variable = { USA_NWM_gdp = gdp_penalty}
    }
}
