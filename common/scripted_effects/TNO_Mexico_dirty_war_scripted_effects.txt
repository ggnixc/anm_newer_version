MEX_setup_DW_connections = {
	# Setup connections to render
	clear_array = MEX_DW_rendered_connections_start
	clear_array = MEX_DW_rendered_connections_end

	for_each_loop = {
		array = MEX_DW_locations
		value = first_loc

		for_each_loop = {
			array = first_loc:MEX_DW_connections
			value = second_loc


			if = {
				limit = { check_variable = { first_loc < second_loc } }
				log = "[?first_loc] [?second_loc]"

				add_to_array = { MEX_DW_rendered_connections_start = first_loc }
				add_to_array = { MEX_DW_rendered_connections_end = second_loc }
			}
		}
	}

	clear_array = MEX_DW_rendered_connections_diff_x
	clear_array = MEX_DW_rendered_connections_diff_y
	clear_array = MEX_DW_rendered_connections_frame
	clear_array = MEX_DW_rendered_connections_geometry_frame
	clear_array = MEX_DW_rendered_connections_f2s
	for_loop_effect = {
		start = 0
		end = MEX_DW_rendered_connections_start^num
		value = i

		set_temp_variable = { first_loc = MEX_DW_rendered_connections_start^i }
		set_temp_variable = { second_loc = MEX_DW_rendered_connections_end^i }

		if = {
			limit = { check_variable = { first_loc:MEX_DW_x > second_loc:MEX_DW_x } }
			set_temp_variable = { connection_x = second_loc:MEX_DW_x }
			set_temp_variable = { difference_x = first_loc:MEX_DW_x }
			subtract_from_temp_variable = { difference_x = second_loc:MEX_DW_x }

			if = {
				limit = { check_variable = { first_loc:MEX_DW_y > second_loc:MEX_DW_y } }
				set_temp_variable = { connection_y = second_loc:MEX_DW_y }
				set_temp_variable = { difference_y = first_loc:MEX_DW_y }
				subtract_from_temp_variable = { difference_y = second_loc:MEX_DW_y }
				set_temp_variable = { toggle = 0 }
				set_temp_variable = { f2s = 0 }
			}
			else = {
				set_temp_variable = { connection_y = first_loc:MEX_DW_y }
				set_temp_variable = { difference_y = second_loc:MEX_DW_y }
				subtract_from_temp_variable = { difference_y = first_loc:MEX_DW_y }
				set_temp_variable = { toggle = 1 }
				set_temp_variable = { f2s = 1 }
			}
		}
		else = {
			set_temp_variable = { connection_x = first_loc:MEX_DW_x }
			set_temp_variable = { difference_x = second_loc:MEX_DW_x }
			subtract_from_temp_variable = { difference_x = first_loc:MEX_DW_x }

			if = {
				limit = { check_variable = { first_loc:MEX_DW_y > second_loc:MEX_DW_y } }
				set_temp_variable = { connection_y = second_loc:MEX_DW_y }
				set_temp_variable = { difference_y = first_loc:MEX_DW_y }
				subtract_from_temp_variable = { difference_y = second_loc:MEX_DW_y }
				set_temp_variable = { toggle = 1 }
				set_temp_variable = { f2s = 0 }
			}
			else = {
				set_temp_variable = { connection_y = first_loc:MEX_DW_y }
				set_temp_variable = { difference_y = second_loc:MEX_DW_y }
				subtract_from_temp_variable = { difference_y = first_loc:MEX_DW_y }
				set_temp_variable = { toggle = 0 }
				set_temp_variable = { f2s = 1 }
			}
		}



		divide_temp_variable = { difference_x = 10 }
		divide_temp_variable = { difference_y = 10 }

		round_temp_variable = difference_x
		round_temp_variable = difference_y



		# Packed frame layout (20 bits total):
		# style * 32768 + diagonal * 16384 + x_length * 64 + y_length
		# x_length is at most 220 and y_length at most 50 on the static map.
		set_temp_variable = { connection_frame = toggle }
		multiply_temp_variable = { connection_frame = 16384 }
		set_temp_variable = { connection_x_component = difference_x }
		multiply_temp_variable = { connection_x_component = 64 }
		add_to_temp_variable = { connection_frame = connection_x_component }
		round_temp_variable = connection_frame
		add_to_temp_variable = { connection_frame = difference_y }

		add_to_array = { MEX_DW_rendered_connections_diff_x = connection_x }
		add_to_array = { MEX_DW_rendered_connections_diff_y = connection_y }
		add_to_array = { MEX_DW_rendered_connections_frame = connection_frame }
		add_to_array = { MEX_DW_rendered_connections_geometry_frame = connection_frame }
		add_to_array = { MEX_DW_rendered_connections_f2s = f2s }
	}
}

# Rebuild only the dynamic style portion of each connection frame. Geometry
# remains static for the lifetime of the Dirty War map.
MEX_DW_refresh_connection_styles = {
	for_loop_effect = {
		start = 0
		end = MEX_DW_rendered_connections_start^num
		value = connection_idx

		set_temp_variable = { start = MEX_DW_rendered_connections_start^connection_idx }
		set_temp_variable = { end = MEX_DW_rendered_connections_end^connection_idx }
		set_temp_variable = { connection_selected = 0 }
		set_temp_variable = { connection_arrow = 0 }
		set_temp_variable = { connection_style = 0 }
		set_temp_variable = { start_group = start:MEX_DW_controller }
		set_temp_variable = { end_group = end:MEX_DW_controller }

		if = {
			limit = {
				OR = {
					check_variable = { start = MEX_DW_selected_location }
					check_variable = { end = MEX_DW_selected_location }
				}
			}
			set_temp_variable = { connection_selected = 1 }
		}

		if = {
			limit = {
				OR = {
					AND = {
						check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 0 }
						check_variable = { end:MEX_DW_parent = start }
					}
					AND = {
						check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 1 }
						check_variable = { start:MEX_DW_parent = end }
					}
				}
			}
			set_temp_variable = { connection_arrow = 1 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 1 }
						check_variable = { end:MEX_DW_parent = start }
					}
					AND = {
						check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 0 }
						check_variable = { start:MEX_DW_parent = end }
					}
				}
			}
			set_temp_variable = { connection_arrow = 2 }
		}

		# Style 1: highlighted path (the first-match rule in the old localisation).
		if = {
			limit = {
				is_in_array = { array = path value = start }
				is_in_array = { array = path value = end }
			}
			set_temp_variable = { connection_style = 1 }
			# Styles 17/18 are consecutive movement edges travelling along or
			# against the shader's p1-to-p2 geometric direction.
			if = {
				limit = {
					OR = {
						check_variable = { end:parent = start }
						check_variable = { start:parent = end }
					}
				}
				if = {
					limit = {
						OR = {
							AND = {
								check_variable = { end:parent = start }
								check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 1 }
							}
							AND = {
								check_variable = { start:parent = end }
								check_variable = { MEX_DW_rendered_connections_f2s^connection_idx = 0 }
							}
						}
					}
					set_temp_variable = { connection_style = 17 }
				}
				else = { set_temp_variable = { connection_style = 18 } }
			}
		}
		else_if = {
			limit = {
				check_variable = { connection_selected = 1 }
				OR = {
					NOT = { MEX_dw_is_connection_allegence_known = yes }
					NOT = { check_variable = { end_group = start_group } }
					check_variable = { start:MEX_DW_controller = 0 }
					check_variable = { end:MEX_DW_controller = 0 }
				}
			}
			set_temp_variable = { connection_style = 2 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { MEX_dw_is_connection_allegence_known = yes }
					NOT = { check_variable = { end_group = start_group } }
					check_variable = { start:MEX_DW_controller = 0 }
					check_variable = { end:MEX_DW_controller = 0 }
					AND = {
						NOT = { is_in_array = { array = MEX_DW_Regions_Focusing value = start:MEX_DW_region } }
						NOT = { is_in_array = { array = MEX_DW_Regions_Focusing value = end:MEX_DW_region } }
						check_variable = { MEX_DW_Regions_Focusing^num > 0 }
					}
				}
			}
			set_temp_variable = { connection_style = 3 }
		}
		else = {
			set_temp_variable = { START_CONTROLLER = start:MEX_DW_controller }
			if = {
				limit = { check_variable = { factional_allignment@var:START_CONTROLLER = 1 } }
				if = {
					limit = { check_variable = { connection_selected = 1 } }
					if = { limit = { check_variable = { connection_arrow = 1 } } set_temp_variable = { connection_style = 4 } }
					else_if = { limit = { check_variable = { connection_arrow = 2 } } set_temp_variable = { connection_style = 5 } }
					else = { set_temp_variable = { connection_style = 6 } }
				}
				else = {
					if = { limit = { check_variable = { connection_arrow = 1 } } set_temp_variable = { connection_style = 7 } }
					else_if = { limit = { check_variable = { connection_arrow = 2 } } set_temp_variable = { connection_style = 8 } }
					else = { set_temp_variable = { connection_style = 9 } }
				}
			}
			else_if = {
				limit = { check_variable = { factional_allignment@var:START_CONTROLLER = 2 } }
				if = {
					limit = { check_variable = { connection_selected = 1 } }
					if = { limit = { check_variable = { connection_arrow = 1 } } set_temp_variable = { connection_style = 10 } }
					else_if = { limit = { check_variable = { connection_arrow = 2 } } set_temp_variable = { connection_style = 11 } }
					else = { set_temp_variable = { connection_style = 12 } }
				}
				else = {
					if = { limit = { check_variable = { connection_arrow = 1 } } set_temp_variable = { connection_style = 13 } }
					else_if = { limit = { check_variable = { connection_arrow = 2 } } set_temp_variable = { connection_style = 14 } }
					else = { set_temp_variable = { connection_style = 15 } }
				}
			}
			else_if = {
				limit = { check_variable = { connection_selected = 1 } }
				set_temp_variable = { connection_style = 16 }
			}
		}

		multiply_temp_variable = { connection_style = 32768 }
		set_temp_variable = { connection_render_frame = MEX_DW_rendered_connections_geometry_frame^connection_idx }
		add_to_temp_variable = { connection_render_frame = connection_style }
		set_variable = { MEX_DW_rendered_connections_frame^connection_idx = connection_render_frame }
	}
}

d_faction = {
	set_temp_variable = { can_we_create_log_entry = 0 }
	set_temp_variable = { faction_controller_changed = 0 }
	set_temp_variable = { requested_faction = args^1 }
	var:args^0 = {
		# Cities are permanently government-controlled. A non-government request
		# is ignored; if older/test state violated the invariant, repair it here.
		if = {
			limit = {
				check_variable = { MEX_DW_node_type = MEX.CITY }
				NOT = { check_variable = { requested_faction = token:MEX_Faction_GOVERNMENT } }
			}
			if = {
				limit = { NOT = { check_variable = { MEX_DW_controller = token:MEX_Faction_GOVERNMENT } } }
				set_variable = { MEX_DW_controller = token:MEX_Faction_GOVERNMENT }
				set_temp_variable = { faction_controller_changed = 1 }
			}
		}
		else = {
			set_variable = { MEX_DW_controller = var:args^1 }
			set_temp_variable = { faction_controller_changed = 1 }

MEX_get_controlled_nodes = {
	#set_temp_variable = { loc_index = 0 }
	for_each_loop = {
		array = MEX_DW_locations
		value = loc

		clear_array = loc:MEX_DW_distance
		clear_array = loc:MEX_DW_supply_through

		set_temp_variable = { mex_dist_index = 0 }
		while_loop_effect = {
			limit = { check_variable = { mex_dist_index < MEX_DW_all_strongholds^num } }

			set_variable = { loc:MEX_DW_distance^mex_dist_index = -1 }
			set_variable = { loc:MEX_DW_supply_through^mex_dist_index = 0 }
			add_to_temp_variable = { mex_dist_index = 1 }
		}

		#set_variable = { loc:MEX_DW_distance^0 = -1 }

	}
	set_temp_variable = { new_stronghold_index = 0 }
	for_each_loop = {
		array = MEX_DW_all_strongholds
		value = stronghold

		set_temp_variable = { group = stronghold:MEX_DW_controller }

		clear_temp_array = to_explore
		clear_array = stronghold:MEX_DW_controlled_nodes
		clear_array = stronghold:MEX_DW_connected_nodes


		add_to_temp_array = { to_explore = stronghold }
		set_variable = { stronghold:MEX_DW_distance^new_stronghold_index = 0 }
		add_to_array = { stronghold:MEX_DW_controlled_nodes = stronghold }

		while_loop_effect = {
			limit = { check_variable = { to_explore^num > 0 } }

			set_temp_variable = { exploring = to_explore^0 }
			remove_from_temp_array = { array = to_explore index = 0 }


			for_each_loop = {
				array = exploring:MEX_DW_connections
				value = exploring_adjacent


				if = {
					limit = { check_variable = { exploring_adjacent:MEX_DW_controller = group } }

					if = {
						limit = { check_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = -1 } }

						set_variable = { exploring_adjacent:MEX_DW_parent^new_stronghold_index = exploring }

						set_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = exploring:MEX_DW_distance^new_stronghold_index }
						add_to_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = 1 }

						add_to_temp_array = { to_explore = exploring_adjacent }
						add_to_array = { stronghold:MEX_DW_controlled_nodes = exploring_adjacent }

					}
				}
				else = {

					if = {
						limit = { check_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = -1 } }
						set_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = exploring:MEX_DW_distance^new_stronghold_index }
						add_to_variable = { exploring_adjacent:MEX_DW_distance^new_stronghold_index = 1 }

						add_to_array = { stronghold:MEX_DW_connected_nodes = exploring_adjacent }

					}

				}
			}
		}
		add_to_temp_variable = { new_stronghold_index = 1 }
	}
	set_variable = { MEX_DW_supply_distance_cache_version = 2 }
}

# Determine the largest remaining movement range in the current selection.
# Only the next, not-yet-running bank is resized, so active banks are untouched.

	set_variable = { MEX_DW_connection_streak_required = 0 }
	for_each_loop = {
		array = selected_regiments
		value = streak_selected_unit
		set_temp_variable = { streak_selected_range = streak_selected_unit:MEX_DW_NODES_TRAVELABLE }
		if = {
			limit = { check_variable = { streak_selected_range > MEX_DW_connection_streak_required } }
			set_variable = { MEX_DW_connection_streak_required = streak_selected_range }
		}
	}
	# The frame layout and route builder currently reserve 32 edge-order values.
	if = {
		limit = { check_variable = { MEX_DW_connection_streak_required > 32 } }
		set_variable = { MEX_DW_connection_streak_required = 32 }
	}
	MEX_DW_ensure_connection_streak_pool = yes

MEX_DW_build_connection_streaks = {

	else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 2 } } set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_2^num } }
	else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 3 } } set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_3^num } }
	else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 4 } } set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_4^num } }
	else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 5 } } set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_5^num } }
	else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 6 } } set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_6^num } }
	else = { set_temp_variable = { streak_bank_capacity = MEX_DW_connection_streak_x_7^num } }

	for_loop_effect = {
		start = 0
		end = streak_bank_capacity
		value = streak_hide_idx
		if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 0 } } set_variable = { MEX_DW_connection_streak_x_0^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_0^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 1 } } set_variable = { MEX_DW_connection_streak_x_1^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_1^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 2 } } set_variable = { MEX_DW_connection_streak_x_2^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_2^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 3 } } set_variable = { MEX_DW_connection_streak_x_3^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_3^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 4 } } set_variable = { MEX_DW_connection_streak_x_4^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_4^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 5 } } set_variable = { MEX_DW_connection_streak_x_5^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_5^streak_hide_idx = -10000 } }
		else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 6 } } set_variable = { MEX_DW_connection_streak_x_6^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_6^streak_hide_idx = -10000 } }
		else = { set_variable = { MEX_DW_connection_streak_x_7^streak_hide_idx = -10000 } set_variable = { MEX_DW_connection_streak_y_7^streak_hide_idx = -10000 } }
	}

	if = {
		limit = { check_variable = { path^num > 1 } }
		set_temp_variable = { streak_edge_count = path^num }
		subtract_from_temp_variable = { streak_edge_count = 1 }
		if = {
			limit = { check_variable = { streak_edge_count > streak_bank_capacity } }
			set_temp_variable = { streak_edge_count = streak_bank_capacity }
		}
		for_loop_effect = {
			start = 0
			end = streak_edge_count
			value = streak_edge_idx

			set_temp_variable = { streak_next_idx = streak_edge_idx }
			add_to_temp_variable = { streak_next_idx = 1 }

			subtract_from_temp_variable = { streak_dx = streak_source:MEX_DW_x }
			set_temp_variable = { streak_dy = streak_destination:MEX_DW_y }
			subtract_from_temp_variable = { streak_dy = streak_source:MEX_DW_y }

			divide_temp_variable = { streak_dy = 10 }
			round_temp_variable = streak_dx
			round_temp_variable = streak_dy

			# obtain the visible source-to-destination traversal order.
			set_temp_variable = { streak_route_order = streak_edge_count }
			subtract_from_temp_variable = { streak_route_order = streak_edge_idx }

			set_temp_variable = { streak_x_component = streak_dx }

			set_temp_variable = { streak_y_component = streak_dy }
			add_to_temp_variable = { streak_y_component = 64 }
			add_to_temp_variable = { streak_frame = streak_y_component }
			round_temp_variable = streak_frame

			if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 0 } } set_variable = { MEX_DW_connection_streak_x_0^streak_edge_idx = streak_source:MEX_DW_x } set_variable = { MEX_DW_connection_streak_y_0^streak_edge_idx = streak_source:MEX_DW_y } set_variable = { MEX_DW_connection_streak_frame_0^streak_edge_idx = streak_frame } }
			else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 1 } } set_variable = { MEX_DW_connection_streak_x_1^streak_edge_idx = streak_source:MEX_DW_x } set_variable = { MEX_DW_connection_streak_y_1^streak_edge_idx = streak_source:MEX_DW_y } set_variable = { MEX_DW_connection_streak_frame_1^streak_edge_idx = streak_frame } }
			else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 2 } } set_variable = { MEX_DW_connection_streak_x_2^streak_edge_idx = streak_source:MEX_DW_x } set_variable = { MEX_DW_connection_streak_y_2^streak_edge_idx = streak_source:MEX_DW_y } set_variable = { MEX_DW_connection_streak_frame_2^streak_edge_idx = streak_frame } }
			else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 3 } } set_variable = { MEX_DW_connection_streak_x_3^streak_edge_idx = streak_source:MEX_DW_x } set_variable = { MEX_DW_connection_streak_y_3^streak_edge_idx = streak_source:MEX_DW_y } set_variable = { MEX_DW_connection_streak_frame_3^streak_edge_idx = streak_frame } }

			else_if = { limit = { check_variable = { MEX_DW_connection_streak_bank = 5 } } set_variable = { MEX_DW_connection_streak_x_5^streak_edge_idx = streak_source:MEX_DW_x } set_variable = { MEX_DW_connection_streak_y_5^streak_edge_idx = streak_source:MEX_DW_y } set_variable = { MEX_DW_connection_streak_frame_5^streak_edge_idx = streak_frame } }

# Unweighted shortest-path search. Dirty War movement range is measured in
# connections, so breadth-first search is both cheaper and more appropriate
# than the former geometric A* approximation. path[] remains stored
# destination-to-source for the movement animation code.
MEX_dw_node_traversal = {
	clear_array = path
	clear_temp_array = MEX_DW_path_queue
	add_to_variable = { MEX_DW_path_search_generation = 1 }

	set_temp_variable = { path_queue_index = 0 }
	set_temp_variable = { path_found = 0 }
	set_variable = { node_A:MEX_DW_path_seen_generation = MEX_DW_path_search_generation }
	set_variable = { node_A:MEX_DW_path_parent = 0 }
	set_variable = { node_A:MEX_DW_path_depth = 0 }
	add_to_temp_array = { MEX_DW_path_queue = node_A }

	while_loop_effect = {
		limit = {
			check_variable = { path_queue_index < MEX_DW_path_queue^num }
			check_variable = { path_found = 0 }
		}

		set_temp_variable = { current_node = MEX_DW_path_queue^path_queue_index }
		add_to_temp_variable = { path_queue_index = 1 }

		if = {
			limit = { check_variable = { current_node = node_B } }
			set_temp_variable = { path_found = 1 }
		}
		else = {
			for_each_loop = {
				array = current_node:MEX_DW_connections
				value = exploring_node

				if = {
					limit = {
						NOT = { check_variable = { exploring_node:MEX_DW_path_seen_generation = MEX_DW_path_search_generation } }
					}
					set_variable = { exploring_node:MEX_DW_path_seen_generation = MEX_DW_path_search_generation }
					set_variable = { exploring_node:MEX_DW_path_parent = current_node }
					set_variable = { exploring_node:MEX_DW_path_depth = current_node:MEX_DW_path_depth }
					add_to_variable = { exploring_node:MEX_DW_path_depth = 1 }
					add_to_temp_array = { MEX_DW_path_queue = exploring_node }
				}
			}
		}
	}

	if = {
		limit = { check_variable = { path_found = 1 } }
		set_temp_variable = { current_node = node_B }
		while_loop_effect = {
			limit = { NOT = { check_variable = { current_node = node_A } } }
			add_to_array = { path = current_node }
			set_temp_variable = { current_node = current_node:MEX_DW_path_parent }
		}
		add_to_array = { path = node_A }
	}
}

# `e MEX_DW_test_selected_gpg_supply = yes`. This deliberately uses the normal
# d_gpg and d_stronghold effects so it tests the real ownership, cache rebuild,

MEX_DW_test_selected_gpg_supply = {
	if = {
		limit = { check_variable = { MEX_DW_selected_location > 0 } }

		# effects; a temporary value is not a safe place to retain the selection.
		set_variable = { MEX_DW_test_supply_stronghold = MEX_DW_selected_location }

		# selected location. This is the same call pattern used by d_gpg/d_faction.
		set_temp_variable = { args^0 = MEX_DW_selected_location }
		set_temp_variable = { args^1 = token:MEX_Faction_THE_GPG }
		d_stronghold = yes

		# Collect the selected node and every node within three connection hops.
		add_to_variable = { MEX_DW_test_supply_generation = 1 }
		clear_temp_array = MEX_DW_test_supply_queue
		set_temp_variable = { test_supply_queue_index = 0 }
		set_variable = { MEX_DW_test_supply_stronghold:MEX_DW_test_supply_seen_generation = MEX_DW_test_supply_generation }
		set_variable = { MEX_DW_test_supply_stronghold:MEX_DW_test_supply_depth = 0 }
		add_to_temp_array = { MEX_DW_test_supply_queue = MEX_DW_test_supply_stronghold }

		while_loop_effect = {
			limit = { check_variable = { test_supply_queue_index < MEX_DW_test_supply_queue^num } }
			set_temp_variable = { test_supply_node = MEX_DW_test_supply_queue^test_supply_queue_index }
			add_to_temp_variable = { test_supply_queue_index = 1 }

			if = {
				limit = { check_variable = { test_supply_node:MEX_DW_test_supply_depth < 3 } }
				for_each_loop = {
					array = test_supply_node:MEX_DW_connections
					value = test_supply_adjacent
					if = {
						limit = {
							NOT = { check_variable = { test_supply_adjacent:MEX_DW_test_supply_seen_generation = MEX_DW_test_supply_generation } }
						}
						set_variable = { test_supply_adjacent:MEX_DW_test_supply_seen_generation = MEX_DW_test_supply_generation }

	else = {

	}
}

MEX_DW_transfer_supplies = {

	}
	set_temp_variable = { stronghold_supply_index = 0 }

		value = stronghold_val

		set_temp_variable = { stronghold_controller = stronghold_val:MEX_DW_controller }

			array = MEX_DW_locations

							check_variable = { finding_cities:MEX_DW_node_type = 1 }

							limit = {
								check_variable = { finding_cities:[GROUP_NAME]_control > 25 }

		}

		clamp_variable = {
			var = stronghold_val:MEX_DW_organization
			max = 100
			min = 0
		}
		#stronghold organization slowly increases its strength
		set_temp_variable = { stronghold_strength_temp = stronghold_val:MEX_DW_organization }
		divide_temp_variable = { stronghold_strength_temp = 5 }
		add_to_variable = { stronghold_val:MEX_DW_Stronghold_Strength = stronghold_strength_temp }
		clamp_variable = {
			var = stronghold_val:MEX_DW_Stronghold_Strength
			max = 100

DW_MEX_modify_recruitment_size = {
	#basically, we attempt to add modif_change (-+10,-+50) to the new amount of recruitable units in a regiment, and then do a ton of checks to see if we have enough equipment to handle said units.
	#if this stops working, dont contact shukla. idk what is happening


	set_temp_variable = { newtotal = recruit_amount@var:MEX_DW_recruit_unit_button_array_value }

	add_to_temp_variable = { newtotal = modif_change }
	clear_array = MEX_DW_Faction_Req_Eq_token_names
	clear_array = MEX_DW_Faction_Req_Eq_token_num
	clear_array = MEX_DW_values_to_inc


	if = {
		limit = {
			check_variable = { newtotal > -1 }
		}


		for_each_loop = {
			array = equipment_required_name@var:MEX_DW_recruit_unit_button_array_value
			value = checkerrrr

			add_to_array = { MEX_DW_Faction_Req_Eq_token_names = checkerrrr }

		}
		for_each_loop = {
			array = equipment_required_num@var:MEX_DW_recruit_unit_button_array_value
			value = checkerrrr


			add_to_array = { MEX_DW_Faction_Req_Eq_token_num = checkerrrr }
		}

		set_temp_variable = { arr_upper = 0 }
		set_temp_variable = { can_continue = 0 }

		while_loop_effect = {
			limit = {
				check_variable = { can_continue = 0 }
				check_variable = { arr_upper < MEX_DW_Faction_Req_Eq_token_names^num }
			}

			set_temp_variable = { equipment_name = MEX_DW_Faction_Req_Eq_token_names^arr_upper }
			set_temp_variable = { temp_dw_equipment_num = MEX_DW_Faction_Req_Eq_token_num^arr_upper }
			multiply_temp_variable = { temp_dw_equipment_num = newtotal }
			meta_effect = {
				text = {
					set_temp_variable = { code_name = token:[EQ_DW_NAME]_code_name }
				}
				EQ_DW_NAME = "[?equipment_name.GetTokenKey]"
			}
			set_temp_variable = { sum_temp_dw_equipment_num = temp_dw_equipment_num }

			for_each_loop = {
				array = MEX_DW_recruit_unit_button_array
				value = units_to_recruit_analysis

				if = {
					limit = {
						check_variable = { recruit_amount@var:units_to_recruit_analysis > 0 }
						NOT = { check_variable = { units_to_recruit_analysis = MEX_DW_recruit_unit_button_array_value } }
					}
					set_temp_variable = { index_of_eq = 0 }
					for_each_loop = {
						array = equipment_required_name@var:units_to_recruit_analysis
						value = find_the_eq_thingy
						break = break_value_thingy

						if = {
							limit = {
								check_variable = { find_the_eq_thingy = equipment_name }
							}
							set_temp_variable = { break_value_thingy = 2 }
						}
						else = {
							add_to_temp_variable = { index_of_eq = 1 }
						}

					}
					meta_effect = {
						text = {
							add_to_temp_variable = { sum_temp_dw_equipment_num = MEX_DW_[ACTUAL_UNIT_NAME_THINGY]_unit_calc_val^index_of_eq }
						}
						ACTUAL_UNIT_NAME_THINGY = "[?units_to_recruit_analysis.GetTokenKey]"
					}
				}

			}
			log = "[?equipment_name.GetTokenLocalizedKey]: [?sum_temp_dw_equipment_num]"

			meta_effect = {
				text = {
					if = {
						limit = {
							check_variable = { sum_temp_dw_equipment_num > num_equipment@[ACTUAL_EQ_NAME]_equipment }
						}
						set_temp_variable = { can_continue = 1 }
					}
					else = {
						add_to_array = { MEX_DW_values_to_inc = temp_dw_equipment_num }

					}
				}
				ACTUAL_EQ_NAME = "[?code_name.GetTokenLocalizedKey]"
			}

			add_to_temp_variable = { arr_upper = 1 }
		}

		if = {
			limit = {
				check_variable = { can_continue = 0 }
			}
			set_variable = { recruit_amount@var:MEX_DW_recruit_unit_button_array_value = newtotal }
			set_temp_variable = { new_upper = 0 }

			for_each_loop = {
				array = MEX_DW_recruit_unit_button_array
				value = can_we_actually_recruit_val
				break = we_can_break

				if = {
					limit = {
						check_variable = { recruit_amount@var:can_we_actually_recruit_val > 0 }
					}
					set_variable = { MEX_DW_Can_Recruit_Unit = 1 }
					set_temp_variable = { we_can_break = 2 }
				}
				else = {
					set_variable = { MEX_DW_Can_Recruit_Unit = 0 }
				}
			}

			meta_effect = {
				text = {
					clear_array = MEX_DW_[ACTUAL_UNIT_NAME]_unit_calc_val
				}
				ACTUAL_UNIT_NAME = "[?MEX_DW_recruit_unit_button_array_value.GetTokenKey]"
			}
			while_loop_effect = {
				limit = {
					check_variable = { new_upper < MEX_DW_Faction_Req_Eq_token_names^num }
				}

				set_temp_variable = { temp_code_value = MEX_DW_Faction_Req_Eq_token_names^new_upper }
				meta_effect = {
					text = {
						add_to_array = { MEX_DW_[ACTUAL_UNIT_NAME]_unit_calc_val = MEX_DW_values_to_inc^new_upper }
					}
					ACTUAL_UNIT_NAME = "[?MEX_DW_recruit_unit_button_array_value.GetTokenKey]"
				}

				add_to_temp_variable = { new_upper = 1 }
			}

		}
	}

	clear_array = MEX_DW_Faction_Req_Eq_token_names
	clear_array = MEX_DW_Faction_Req_Eq_token_num
	clear_array = MEX_DW_values_to_inc
}

		add_to_array = { MEX_DW_Bonuses = bonus }
	}
	else = {
		add_to_array = { Active_Bonuses@var:faction_active = bonus }
		remove_from_array = { MEX_DW_Bonuses = bonus }
		#add to front
		add_to_array = {
			array = MEX_DW_Bonuses
			value = bonus
			index = 0
		}
		#decide the image
		if = {
			limit = {
				check_variable = { Active_group@var:bonus = -1 }
			}
			set_temp_variable = { adddderrrr = factional_allignment@var:faction_active }
			add_to_temp_variable = { adddderrrr = 1 }
			set_variable = { Active_group@var:bonus = adddderrrr }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { Active_group@var:bonus = 1 }
						check_variable = { factional_allignment@var:faction_active = 1 }
					}
					AND = {
						check_variable = { Active_group@var:bonus = 2 }
						check_variable = { factional_allignment@var:faction_active = 0 }
					}
				}
			}
			set_variable = { Active_group@var:bonus = 4 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { Active_group@var:bonus = 2 }
						check_variable = { factional_allignment@var:faction_active = 2 }
					}
					AND = {
						check_variable = { Active_group@var:bonus = 3 }
						check_variable = { factional_allignment@var:faction_active = 1 }
					}
				}
			}
			set_variable = { Active_group@var:bonus = 5 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { Active_group@var:bonus = 0 }
						check_variable = { factional_allignment@var:faction_active = 2 }
					}
					AND = {
						check_variable = { Active_group@var:bonus = 3 }
						check_variable = { factional_allignment@var:faction_active = 0 }
					}
				}
			}
			set_variable = { Active_group@var:bonus = 6 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { Active_group@var:bonus = 3 }
					check_variable = { Active_group@var:bonus = 4 }
					check_variable = { Active_group@var:bonus = 5 }
				}
			}
			set_variable = { Active_group@var:bonus = 7 }
		}
	}
}

MEX_DW_Should_Battle_Start = {

	if = {
		limit = {
			check_variable = { node_To:MEX_DW_node_units^num > 1 }
			check_variable = { node_To:MEX_DW_intel > 25 } #arb
			check_variable = { node_To:Is_In_Battle = 0 } #dont want a battle to keep restarting
		}

		set_temp_variable = { first_unit = node_To:MEX_DW_node_units^0 }
		set_temp_variable = { first_unit_alliegance = first_unit:MEX_DW_alliegance }

		for_each_loop = {
			array = node_To:MEX_DW_node_units
			value = checker_unit_val
			break = break

			if = {
				limit = {
					NOT = {
						check_variable = { checker_unit_val:MEX_DW_alliegance = first_unit_alliegance }
					}
				}
				set_variable = { node_To:MEX_DW_Battle_First_Group = first_unit_alliegance }
				set_variable = { node_To:MEX_DW_Battle_Second_Group = checker_unit_val:MEX_DW_alliegance }
				set_variable = { node_To:Is_In_Battle = 1 }
				set_temp_variable = { break = 1 }
			}
		}
		if = {
			limit = {
				check_variable = { node_To:Is_In_Battle = 1 }
			}
			MEX_DW_Organize_New_Battle = yes

			if = {
				limit = {
					check_variable = { node_To:Node_Not_Battle^num > 0 }
				}
				clear_array = node_To:Node_Not_Battle
				MEX_DW_update_gui_elements = yes
			}
		}
	}
}
MEX_DW_Organize_New_Battle = {
	add_to_array = { MEX_DW_Ongoing_Battles = node_To }

	for_each_loop = {
		array = node_To:MEX_DW_node_units
		value = unit_value

		log = "checking: [?unit_value]"

		if = {
			limit = {
				check_variable = { unit_value:MEX_DW_alliegance = node_To:MEX_DW_Battle_First_Group }
			}
			add_to_array = { node_To:MEX_DW_First_Group_Units = unit_value }
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = node_To:MEX_DW_First_Group_Unit_Types
							value = unit_value:MEX_DW_UNIT_TYPE
						}
					}
				}
				add_to_array = { node_To:MEX_DW_First_Group_Unit_Types = unit_value:MEX_DW_UNIT_TYPE }
			}
			
		}
		else_if = {
			limit = {
				check_variable = { unit_value:MEX_DW_alliegance = node_To:MEX_DW_Battle_Second_Group }
			}
			add_to_array = { node_To:MEX_DW_Second_Group_Units = unit_value }
			if = {
				limit = {
					NOT = {
						is_in_array = {
							array = node_To:MEX_DW_Second_Group_Unit_Types
							value = unit_value:MEX_DW_UNIT_TYPE
						}
					}
				}
				add_to_array = { node_To:MEX_DW_Second_Group_Unit_Types = unit_value:MEX_DW_UNIT_TYPE }
			}
		}
	}

	for_each_loop = { 
		array = node_To:MEX_DW_First_Group_Units
		value = u_valone

		add_to_temp_variable = { unitone_sum = u_valone:MEX_DW_node_num_units }
	}
	for_each_loop = { 
		array = node_To:MEX_DW_Second_Group_Units
		value = u_valone

		add_to_temp_variable = { unitwo_sum = u_valone:MEX_DW_node_num_units }
	}
	subtract_from_temp_variable = { unitone_sum = unitwo_sum }
	multiply_temp_variable = { unitone_sum = -0.01 }
	set_temp_variable = { input = 2.718 }
	set_temp_variable = { exp = unitone_sum }
	TNO_arith_exponential = yes
	add_to_temp_variable = { input = 1 }
	set_variable = { node_To:MEX_DW_Battle_leans = 50 }
	divide_variable = { node_To:MEX_DW_Battle_leans = input }
	add_to_variable = { node_To:MEX_DW_Battle_leans = 25 }
	#need to remember to recall MEX_DW_Should_Battle_Start at the end of the battle in case another faction walks in while your fighting another faction
}

	set_temp_variable = { yB = node_2:MEX_DW_y }

	subtract_from_temp_variable = { yDiff = yA }
	divide_temp_variable = { xDiff = 100 }
	divide_temp_variable = { yDiff = 100 }
	multiply_temp_variable = { xDiff = xDiff }

	multiply_temp_variable = { yDiff = yDiff }

	set_temp_variable = { dist_sqrd = 0 }
	add_to_temp_variable = { dist_sqrd = xDiff }
	add_to_temp_variable = { dist_sqrd = yDiff }

	set_temp_variable = { eucdist = dist_sqrd }
}
