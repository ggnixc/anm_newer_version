TNO_add_assorted_law = {
	add_to_array = { TNO_assorted_laws = temp_law }
	custom_effect_tooltip = TNO_add_assorted_law_tt
}

TNO_remove_assorted_law = {
	remove_from_array = { TNO_assorted_laws = temp_law }
	custom_effect_tooltip = TNO_remove_assorted_law_tt
}
