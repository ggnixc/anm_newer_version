misc_law_array_startup = {
	clear_array = TNO_assorted_laws_display

	add_to_array = { TNO_assorted_laws_display = 1 }
	add_to_array = { TNO_assorted_laws_display = 2 }
	add_to_array = { TNO_assorted_laws_display = 3 }
	add_to_array = { TNO_assorted_laws_display = 4 }
	add_to_array = { TNO_assorted_laws_display = 5 }
	add_to_array = { TNO_assorted_laws_display = 6 }
	add_to_array = { TNO_assorted_laws_display = 7 }
	add_to_array = { TNO_assorted_laws_display = 8 }
}

misc_law_array_fix_scrollbar = {
	while_loop_effect = {
		limit = {
			is_in_array = { TNO_assorted_laws_display = -1 }
		}
		remove_from_array = { TNO_assorted_laws_display = -1 }
	}
	
	set_temp_variable = { temp_modulo = TNO_assorted_laws_display^num }
	modulo_temp_variable = { temp_modulo = 2 }

	if = {
		limit = { check_variable = { temp_modulo = 1 } }
		add_to_array = { TNO_assorted_laws_display = -1 }
	}
	add_to_array = { TNO_assorted_laws_display = -1 }
	add_to_array = { TNO_assorted_laws_display = -1 }
}
