EW_Button_Large_click = { 
	if = { 
		limit = { NOT = { has_country_flag = GER_EW_Decisions_GUI_Tab1 } }
		set_country_flag = GER_EW_Decisions_GUI_Tab1
		clr_country_flag = GER_EW_Decisions_GUI_Tab2
		clr_country_flag = GER_EW_Continent
		}
	else_if = { 
		limit = { has_country_flag = GER_EW_Decisions_GUI_Tab1 }
		clr_country_flag = GER_EW_Decisions_GUI_Tab1
		clr_country_flag = GER_EW_Country
		set_country_flag = GER_EW_Decisions_GUI_Tab2

	}
}
