TNO_fopo_should_display_homefront = {
	OR = {
		AND = {
			tag = USA
			always = no
		}
		AND = {
			tag = GER
			always = no
		}
		AND = {
			tag = JAP
			always = no
		}
	}
}
TNO_fopo_show_info_tab = {
	OR = {
		AND = {
			tag = USA
			check_variable = { selected_conflict_id = token:GC_WAC }
		}
	}
}

ANM_fopo_finland_selected = {
	OR = {
		check_variable = { selected_conflict_id = token:GC_FinlandCW }
		AND = {
			is_in_array = { TNO_active_conflicts = token:GC_FinlandCW }
			is_ai = yes
		}
	}
}
