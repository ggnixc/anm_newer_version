## Germany-America
Cold_War_GUI_GER_USA_Relations_Max = {
	custom_trigger_tooltip = {
		tooltip = TNO_Cold_War_GUI_GER_USA_relations_at_max_tt
		check_variable = { global.ColdWarRelations_GER_USA = 10 }
	}
}
Cold_War_GUI_GER_USA_Relations_Min = {
	custom_trigger_tooltip = {
		tooltip = TNO_Cold_War_GUI_GER_USA_relations_at_min_tt
		check_variable = { global.ColdWarRelations_GER_USA = -10 }
	}
}
Cold_War_GUI_GER_USA_Relations_NotMax = {
	custom_trigger_tooltip = {
		tooltip = TNO_Cold_War_GUI_GER_USA_relations_not_already_at_max_tt
		NOT = { check_variable = { global.ColdWarRelations_GER_USA = 10 } }
	}
}
Cold_War_GUI_GER_USA_Relations_AtLeast = { # Uses N temp variable
	custom_trigger_tooltip = {
		tooltip = TNO_Cold_War_GUI_GER_USA_relations_val_tt
		NOT = { check_variable = { global.ColdWarRelations_GER_USA < N } }
	}
}
