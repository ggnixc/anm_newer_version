#Scripted effects for regionalizing world events are stored here, I call them "interest effects" - Cringe Larper
#Figured I should explain here how these actually work:
#these effects should be added inside the "limit = {}" before the code to fire a news event, alongside "is_ai = no" since the AI should never get them (this is also the reason news events should NOT have effects in them btw!)
#This system should NOT be applied to every single event; events of world wide importance, such as the Aleutian Crisis, Oil Crisis civil wars or Superpower election/regime change are exempt from it
#While continent based interest effects should be used for most events, since many tags have inherent interest in other tags/region (ex. Germany with Pakistan), more specific interest effects are used instead (so a news event coming from Pakistan would use HasGermanAsiaInterest instead of HasAsiaInterest)
#More interest effects can be added as needed
#As a general rule, if there's a news event coming from a specific tag and there exist an interest effect that specifically covers that tag or that tag's region, use it; if not just use the effect relating to that tag's continent
#Follow the comments besides each effect for guidance
#If needed, you can also use multiple interest effects inside a single "Limit = {}," just remember to add an "OR = {}" first (don't put "is_ai = no" inside the "OR = {}" too though!)

########################
#######  EUROPE  #######
########################

HasEuropeInterest = {
	OR = {
		capital_scope = { is_in_europe = yes }
		capital_scope = { is_in_antarctica = yes }
		is_in_faction_with = GER
		check_variable = { ECON_SPHERE_LEADER = ITA }
		is_russian_nation = yes
		original_tag = TUR
		has_country_flag = Custom_Europe_Interest
	}
}

HasIberiaInterest = {
	OR = {
		HasEuropeInterest = yes
		capital_scope = { is_in_south_america = yes }
		original_tag = MEX
	}
}

HasIberianWarsInterest = {
	OR = {
		HasIberiaInterest = yes
		capital_scope = { is_in_north_america = yes }
	}
}

HasItalyInterest = {
	OR = {
		HasEuropeInterest = yes
		original_tag = ARG
		original_tag = JAP
	}
}

HasBritishInterest = { #For free Britain. Also (OFN alligned) Ireland, Iceland and Faroe
	OR = {
		HasEuropeInterest = yes
		is_in_faction_with = USA
	}
}

	OR = {

	}

	OR = {
		HasAsiaInterest = yes
		HasEuropeInterest = yes
	}
}

	}
}

HasSouthAsiaInterest = { #India and any OFN alligned nation in south Asia OR any time there's a Sphere-OFN proxy going on (indonesia, Philippines, etc) (if the OFN loses the proxy and cannot influence the tag further then stop using this for that tag)
