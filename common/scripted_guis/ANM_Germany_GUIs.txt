	ANM_reichsstability_Button_GUI = {
		context_type = player_context
		window_name = "ANM_reichsstability_Button_GUI"
		parent_window_token = politics_tab

		visible = {
			tag = GER
		}
		
		triggers = {
		}
		
		effects = {
			reichsstability_open_click = {
				if = {
					limit = { NOT = { has_country_flag = GER_show_reichsstability_window } }
					set_country_flag = GER_show_reichsstability_window
				}
				else_if = {
					limit = { has_country_flag = GER_show_reichsstability_window }
					clr_country_flag = GER_show_reichsstability_window
				}
			}
		}
		
		properties = {
		}
	}
	ANM_reichsstabilitat_GUI = {
		context_type = player_context
		window_name = "ANM_reichsstabilitat_GUI"
		parent_window_token = "ANM_reichsstability_Button_GUI"
		visible = {
			has_country_flag = GER_show_reichsstability_window
		}
		
		triggers = {
			reichsstability_party_button_2_click_enabled = {
				if = {
					limit = {has_country_flag = reichsstability_tier_2}
					AND ={
						NOT = { has_country_flag = GER_reichsstabilitat_interaction_1 }
						has_political_power > 50
					}
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_3}
					AND ={
						NOT = { has_country_flag = GER_reichsstabilitat_interaction_2 }
						has_political_power > 50
					}
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_4}
					AND ={
						NOT = { has_country_flag = GER_reichsstabilitat_interaction_3 }
						has_political_power > 50
					}
				}
				else = {
					always = no
				}
			}
			reichsstability_party_button_1_click_enabled = {
				if = {
					limit = {has_country_flag = reichsstability_tier_2}
					NOT = { has_country_flag = GER_reichsstabilitat_interaction_1 }
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_3}
					NOT = { has_country_flag = GER_reichsstabilitat_interaction_2 }
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_4}
					NOT = { has_country_flag = GER_reichsstabilitat_interaction_3 }
				}
				else = {
					always = no
				}
				
			}
			reichsstability_tier_button_1_click_enabled = {
				NOT = 
				{
					has_country_flag = {
						flag = reichsstability_tier_1
					}
				}
			}
			reichsstability_tier_button_2_click_enabled = {
				NOT = 
				{
					has_country_flag = {
						flag = reichsstability_tier_2
					}
				}
			
			}
			reichsstability_tier_button_3_click_enabled = {
				NOT = 
				{
					has_country_flag = {
						flag = reichsstability_tier_3
					}
				}
			
			}
			reichsstability_tier_button_4_click_enabled = {
				NOT = 
				{
					has_country_flag = {
						flag = reichsstability_tier_4
					}
				}
			}
		}
		
		effects = {
			reichsstabilitat_close_click = {
				clr_country_flag = GER_show_reichsstability_window
			}
			
			reichsstability_party_button_2_click = {
				if = {
					limit = {has_country_flag = reichsstability_tier_2}
					if = {
						limit = { has_political_power > 50 }
						GER = {add_political_power = -50}
						GER_speer_influence_top = yes
						GER_power_struggle_pulse_top = yes
						set_country_flag = {
							flag = GER_reichsstabilitat_interaction_1
							value = 1
							days = 30
						}
					}
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_3}
					if = {
						limit = { has_political_power > 50 }
						GER = {add_political_power = -50}
						GER_speer_influence_middle = yes
						GER_power_struggle_pulse_middle = yes
						set_country_flag = {
							flag = GER_reichsstabilitat_interaction_2
							value = 1
							days = 30
						}
					}
				}
				else = {
					if = {
						limit = { has_political_power > 50 }
						GER = {add_political_power = -50}
						GER_speer_influence_bottom = yes
						GER_power_struggle_pulse_bottom = yes
						set_country_flag = {
							flag = GER_reichsstabilitat_interaction_3
							value = 1
							days = 30
						}
					}
				}
					
				
			}
			reichsstability_party_button_1_click = {
				if = {
					limit = {has_country_flag = reichsstability_tier_2}
					GER = {add_political_power = 50}
					GER_front_influence_top = yes
					GER_power_struggle_pulse_top = yes
					set_country_flag = {
						flag = GER_reichsstabilitat_interaction_1
						value = 1
						days = 30
					}
				}
				else_if = {
					limit = {has_country_flag = reichsstability_tier_3}
					GER = {add_political_power = 50}
					GER_front_influence_middle = yes
					GER_power_struggle_pulse_middle = yes
					flag = GER_reichsstabilitat_interaction_2
					set_country_flag = {
						flag = GER_reichsstabilitat_interaction_2
						value = 1
						days = 30
					}
				}
				else = {
					GER = {add_political_power = 50}
					GER_front_influence_bottom = yes
					GER_power_struggle_pulse_bottom = yes
					flag = GER_reichsstabilitat_interaction_3
					set_country_flag = {
						flag = GER_reichsstabilitat_interaction_3
						value = 1
						days = 30
					}
				}
			}
			reichsstability_tier_button_1_click = {
				set_country_flag = reichsstability_tier_1
				clr_country_flag = reichsstability_tier_4
				clr_country_flag = reichsstability_tier_2
				clr_country_flag = reichsstability_tier_3
				GER_power_struggle_pulse_peak = yes
			}
			reichsstability_tier_button_2_click = {
				set_country_flag = reichsstability_tier_2
				clr_country_flag = reichsstability_tier_1
				clr_country_flag = reichsstability_tier_4
				clr_country_flag = reichsstability_tier_3
				GER_power_struggle_pulse_top = yes
			}
			reichsstability_tier_button_3_click = {
				set_country_flag = reichsstability_tier_3
				clr_country_flag = reichsstability_tier_1
				clr_country_flag = reichsstability_tier_2
				clr_country_flag = reichsstability_tier_4
				GER_power_struggle_pulse_middle = yes
			}
			reichsstability_tier_button_4_click = {
				set_country_flag = reichsstability_tier_4
				clr_country_flag = reichsstability_tier_1
				clr_country_flag = reichsstability_tier_2
				clr_country_flag = reichsstability_tier_3
				GER_power_struggle_pulse_bottom = yes
			}
			
		}
		
		properties = {
			reichsstabilitat_bar_opposition = { frame = GER_progbar_frame }
		}
	}
