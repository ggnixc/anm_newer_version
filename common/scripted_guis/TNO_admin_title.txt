scripted_gui = {
	TNO_AdminTitle_GUI = {
		window_name = "TNO_AdminTitle_GUI"
		context_type = selected_state_context
		parent_window_token = selected_state_view

		visible = {	
			OR = {
				has_variable = AdminTitle
				has_variable = AssocRegion
				has_variable = SubAdminTitle
			}
			NOT = { has_global_flag = TNO_my_world_is_on_fire_how_about_yours }
		}

		effects = {
			AdminTitle_Button_click = {
				ROOT = {
					set_variable = { TNO_admin_title_map_mode_selected_state = PREV }
					clr_country_flag = TNO_admin_title_map_mode_same_country_only
					force_update_map_mode = { mapmode = tno_admin_title_map_mode }
				}
			}
			AdminTitle_Button_right_click = {
				ROOT = {
					set_variable = { TNO_admin_title_map_mode_selected_state = PREV }
					PREV.owner = { save_global_event_target_as = TNO_admin_title_map_mode_selected_state_owner }
					set_country_flag = TNO_admin_title_map_mode_same_country_only
					force_update_map_mode = { mapmode = tno_admin_title_map_mode }
				}
			}
			AdminTitle_Button_shift_click = {
				ROOT = {
					if = {
						limit = {
							NOT = { has_country_flag = TNO_admin_title_map_mode_info_display }
						}
						set_country_flag = TNO_admin_title_map_mode_info_display
					}
					else = {
						clr_country_flag = TNO_admin_title_map_mode_info_display
					}
				}
			}
		}

		## Ukraine Breadbasket
		triggers = {
			breadbasket_icon_visible = {
				controller = {
					has_country_flag = UKR_bb_enabled
					is_ai = no
				}
			}
			china_provinces_short_visible = {
				controller = {
					OR = {
						tag = CHI
						is_in_array = { global.chinese_warlords = THIS }
					}
				}
			}
		}
		properties = {
			china_provinces_short = {
				frame = THIS.china_state_ref_GUI
			}
		}
	}

	TNO_AdminTitle_MapmodeControlGUI = {
		window_name = "AdminTitle_MapmodeControlInterface"
		context_type = player_context
		parent_window_name = "WindowWithAllMapModes_Standalone"
		mapmode = tno_admin_title_map_mode

		visible = { always = yes }

		triggers = {
			admin_title_button_click_enabled = {
				NOT = { check_variable = { TNO_admin_title_map_mode_config = 1 } }
			}
			primary_assoc_region_button_click_enabled = {
				NOT = { check_variable = { TNO_admin_title_map_mode_config = 2 } }
			}
			secondary_assoc_region_button_click_enabled = {
				NOT = { check_variable = { TNO_admin_title_map_mode_config = 3 } }
			}
			sub_admin_title_button_click_enabled = {
				NOT = { check_variable = { TNO_admin_title_map_mode_config = 4 } }
			}
		}

		effects = {
			admin_title_button_click = {
				set_variable = { TNO_admin_title_map_mode_config = 1 }
				force_update_map_mode = { mapmode = tno_admin_title_map_mode }
			}
			primary_assoc_region_button_click = {
				set_variable = { TNO_admin_title_map_mode_config = 2 }
				force_update_map_mode = { mapmode = tno_admin_title_map_mode }
			}
			secondary_assoc_region_button_click = {
				set_variable = { TNO_admin_title_map_mode_config = 3 }
				force_update_map_mode = { mapmode = tno_admin_title_map_mode }
			}
			sub_admin_title_button_click = {
				set_variable = { TNO_admin_title_map_mode_config = 4 }
				force_update_map_mode = { mapmode = tno_admin_title_map_mode }
			}
		}
	}
