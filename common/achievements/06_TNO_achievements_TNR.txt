RUS_All_Russian_Saviour = {
	possible = {
		has_start_date < 1962.01.02
		tag = AMR
	}

	happened = {
		custom_trigger_tooltip = {
			AND = {
				has_idea = AMR_Altar_Of_Fascism
				check_variable = { AMR_JAP_support > 70 }
				check_variable = { var = AMR_fascist_acceptance value = 90 compare = greater_than_or_equals }
				check_variable = { var = AMR_local_disaffection value = 50 compare = less_than_or_equals }
				has_country_flag = AMR_BolotovChain
			}
			tooltip = RUS_All_Russian_Saviour_trigger
		}
	}
}
