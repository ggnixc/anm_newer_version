add_namespace = MOR_skeleton_event

country_event = {
	id = MOR_skeleton_event.1
	title = MOR_skeleton_event.1.t
	desc = MOR_skeleton_eventevent.1.desc

	fire_only_once = yes

	option = { 
		name = MOR_skeleton_event.1.a
		if = { 
			limit = { has_dlc = "Together for Victory" }
			MAR = { 
				get_current_government_type = yes
				save_party_popularities = yes	
			}
			MOR = { 
				set_autonomy = { 
					target = MAR 
					autonomy_state = autonomy_colony
				} 
			} 
			MAR = {
				restore_previous_government_type = yes
				apply_party_popularities = yes
			}

			else = {
				MAR = { 
					get_current_government_type = yes
					save_party_popularities = yes	
				}
				MOR = { puppet = MAR } 
				MAR = {
					restore_previous_government_type = yes
					apply_party_popularities = yes
				}
			}
		} 

	}
}
